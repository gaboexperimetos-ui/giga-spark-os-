# Gigaspark OS - native SDL2 simulator
cmake_minimum_required(VERSION 3.20)
project(gigaspark_sim C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# Prefer SDL2's installed CMake package; fall back to Linux pkg-config.
find_package(SDL2 CONFIG QUIET)
if(TARGET SDL2::SDL2)
  set(GIGA_SDL_TARGET SDL2::SDL2)
elseif(TARGET SDL2::SDL2-static)
  set(GIGA_SDL_TARGET SDL2::SDL2-static)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  set(GIGA_SDL_TARGET PkgConfig::SDL2)
endif()

add_executable(gigaspark_sim main.c)
target_include_directories(gigaspark_sim PRIVATE ../common ../sdk ../common/include)
target_link_libraries(gigaspark_sim PRIVATE ${GIGA_SDL_TARGET} m)
if(TARGET SDL2::SDLmain)
  target_link_libraries(gigaspark_sim PRIVATE SDL2::SDLmain)
elseif(TARGET SDL2::SDL2main)
  target_link_libraries(gigaspark_sim PRIVATE SDL2::SDL2main)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(gigaspark_sim PRIVATE -Wall -Wextra -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME simulator_smoke COMMAND gigaspark_sim --smoke)
  set_tests_properties(simulator_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
endif()
install(TARGETS gigaspark_sim RUNTIME DESTINATION bin)
